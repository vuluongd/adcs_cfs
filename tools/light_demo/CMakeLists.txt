cmake_minimum_required(VERSION 3.16)
project(light_demo C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Werror -pedantic)
set(LIGHT_DIR "${CMAKE_CURRENT_LIST_DIR}/../../apps/light_app")
add_library(light_core STATIC "${LIGHT_DIR}/fsw/src/light_tracking.c" "${LIGHT_DIR}/fsw/src/light_sensor.c"
  "${LIGHT_DIR}/fsw/src/light_angle.c" "${LIGHT_DIR}/fsw/src/light_sim.c"
  "${LIGHT_DIR}/fsw/src/light_pins.c")
target_include_directories(light_core PUBLIC "${LIGHT_DIR}/fsw/inc" "${LIGHT_DIR}/fsw/src")
target_link_libraries(light_core PUBLIC m)
add_executable(light_demo main.c)
target_link_libraries(light_demo PRIVATE light_core)
enable_testing()
add_executable(test_light "${LIGHT_DIR}/tests/test_light.c")
target_link_libraries(test_light PRIVATE light_core)
add_test(NAME bearing_and_validity COMMAND test_light)
set_tests_properties(bearing_and_validity PROPERTIES ENVIRONMENT "LIGHT_APP_BACKEND=sim")
add_executable(test_sensor "${LIGHT_DIR}/tests/test_sensor.c")
target_link_libraries(test_sensor PRIVATE light_core)
target_link_options(test_sensor PRIVATE
  -Wl,--wrap=open,--wrap=ioctl,--wrap=write,--wrap=read,--wrap=close,--wrap=clock_gettime)
add_test(NAME i2c_protocol_and_recovery COMMAND test_sensor)
add_executable(test_adcs_receiver "${LIGHT_DIR}/tests/test_adcs_receiver.c")
target_link_libraries(test_adcs_receiver PRIVATE light_core)
add_test(NAME adcs_freshness_and_validation COMMAND test_adcs_receiver)
add_executable(test_modules "${LIGHT_DIR}/tests/test_modules.c")
target_link_libraries(test_modules PRIVATE light_core)
add_test(NAME angle_simulation_and_pins COMMAND test_modules)
